    @Column //REVIEW: switch all fields to private use getter to access the fields

    @Column //REVIEW: set @min @max to have boundaries for the values
    public int stock;
//REVIEW: risk of fetching all records donot use CascadeType.ALL with eager fetch
//REVIEW: have @equals @hashcode methods

// Controller
@RestController
class InventoryController {

//REVIEW: do bean injection here instead of new
    private InventorySystem system = new InventorySystem();
//REVIEW: use response entity instead of string
    @GetMapping("/updateStock")  //REVIEW: do @postMapping + add @valid
    public String updateStock(@RequestParam String itemId, @RequestParam int qty) {
        system.updateStock(itemId, qty);
        return "Stock updated";
    }

    @GetMapping("/items") //REVIEW: do pagination take page size + use response entity exposing DTO structure ere avoid it
    public List<Item> items() {
        return system.getAllItems();
    }

    @GetMapping("/export")  //REVIEW: return object need to be responseentity + use status code to communiate success or failure
    public String export() {
        system.exportData();
        return "Export complete";
    }
}
